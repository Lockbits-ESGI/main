\begin{frame}{Contexte \& Objectifs}
  \begin{block}{Contexte}
    Entreprise fictive de cybersécurité souhaitant se doter d'une infrastructure
    complète de sniffing et d'analyse de menaces.
  \end{block}

  \begin{block}{4 composants majeurs}
    \begin{itemize}
      \bitem \textbf{EDR} — Agent de détection déployé sur les postes clients + Serveur central
      \bitem \textbf{Site client} — Portail client avec suivi des alertes et tickets GLPI
      \bitem \textbf{Chatbot} — Assistant automatique alimenté par l'API Claude
      \bitem \textbf{DevOps} — CI/CD automatisé, conteneurisation Docker, monitoring complet
    \end{itemize}
  \end{block}

  \textbf{Objectif :} livrer une infrastructure professionnelle de A à Z, de
  l'écriture du code au déploiement en production.
\end{frame}

\begin{frame}{Architecture Globale}

    \begin{itemize}
      \bitem Agent $\rightarrow$ EDR $\rightarrow$ VirusTotal / GLPI
      \bitem Client $\rightarrow$ Site $\rightarrow$ MySQL / GLPI
      \bitem Chatbot $\rightarrow$ Claude API
    \end{itemize}
  \end{columns}

  \vspace{0.5em}
  \begin{alertblock}{Organisation du code}

% Slide 4 — Composants Applicatifs

\begin{frame}{Composants Applicatifs}
  \begin{description}
    \item[\textbf{EDR Server}] \hfill \\
      FastAPI, API REST, détection de menaces, VirusTotal,
      création automatique de tickets GLPI, métriques Prometheus
    \item[\textbf{EDR Agent}] \hfill \\
      Python, packagé via PyInstaller, scan fichiers / monitoring /
      FIM (\textit{File Integrity Monitoring}), heartbeat, déploiement one-liner
    \item[\textbf{Site Client}] \hfill \\
      PHP 8+, Apache, MySQL, Tailwind CSS, SSO GLPI
    \item[\textbf{Chatbot}] \hfill \\
      Proxy PHP vers Claude API, rate limiting (100 requêtes/jour)
  \end{description}
\end{frame}

% ============================================================
% Slide 5 — Infrastructure \& CI/CD
% ============================================================
\begin{frame}{Infrastructure \& CI/CD}
  \textbf{Conteneurisation}
  \begin{itemize}
    \bitem Docker multi-architecture (\texttt{amd64} + \texttt{arm64})
    \bitem Docker Compose : environnements \texttt{dev}, \texttt{preprod}, \texttt{prod}
    \bitem Registry GHCR (\textit{GitHub Container Registry})
  \end{itemize}

  \textbf{CI/CD 100\% homemade}
  \begin{itemize}
    \bitem Pipeline : push $\rightarrow$ lint $\rightarrow$ test $\rightarrow$ build $\rightarrow$ scan Trivy $\rightarrow$ push GHCR $\rightarrow$ webhook Dokploy $\rightarrow$ déploiement en $\sim$5\,min
    \bitem Stratégie : \texttt{master} $\rightarrow$ prod (\texttt{:latest}), \texttt{develop} $\rightarrow$ preprod (\texttt{:develop})
  \end{itemize}

  \textbf{Hébergement \& Monitoring}
  \begin{itemize}
    \bitem Proxmox personnel, LXC Dokploy, LXC cloudflared, Cloudflare Zero Trust
    \bitem Stack observabilité : Prometheus, Grafana, Loki, Alloy
  \end{itemize}
\end{frame}
