% --- Slide 5 : CI/CD Pipeline ---
\begin{frame}{CI/CD — Pipeline 100\% Homemade}
  \textbf{Pipeline complète (GitHub Actions)}
  \begin{enumerate}
    \item \textbf{Push} sur \texttt{develop} ou \texttt{main}
    \item \textbf{Lint} : Ruff (Python), PHP CS Fixer
    \item \textbf{Tests unitaires} : pytest (EDR), PHPUnit (Site)
    \item \textbf{Build multi-arch} : Docker buildx (amd64 + arm64)
    \item \textbf{Push} vers GitHub Container Registry (GHCR)
    \item \textbf{Scan Trivy} : blocage si CRITICAL ou HIGH
    \item \textbf{Webhook} $\rightarrow$ Dokploy $\rightarrow$ déploiement auto
  \end{enumerate}
  \vspace{0.3cm}
  \textbf{Stratégie}
  \begin{itemize}
    \bitem \texttt{main} $\rightarrow$ \texttt{:latest} $\rightarrow$ \textbf{production}
    \bitem \texttt{develop} $\rightarrow$ \texttt{:develop} $\rightarrow$ \textbf{pré-production}
  \end{itemize}
  \begin{exampleblock}{$\sim$5 minutes du push au déploiement — zéro intervention manuelle}
  \end{exampleblock}
\end{frame}
\note{
  Insister sur le fait que tout est fait maison — pas de templates GitHub Actions pré-construits.
  Le pipeline tourne en \textasciitilde5 minutes : le temps de build multi-arch est le plus long.
  Le scan Trivy est bloquant : si une vulnérabilité CRITICAL est trouvée, l'image n'est pas déployée.
  Déploiement via webhook : Dokploy écoute les webhooks et redéploie automatiquement.
}

\section{Swann — EDR Server \& Agent}

% --- Slide 8 : EDR Server Architecture ---
\begin{frame}{EDR Server — Architecture}
  \textbf{Un serveur FastAPI (Python) conçu pour la détection}
  \begin{itemize}
    \bitem API REST asynchrone avec Uvicorn
    \bitem \textbf{Endpoints} : \texttt{/health}, \texttt{/api/v1/stats}, \texttt{/api/v1/agents}, \texttt{/api/v1/events}
    \bitem Authentification par \textbf{API key} partagée avec l'agent
    \bitem \textbf{Rate limiting} : protection contre les abus
    \bitem \textbf{CORS} : configurable pour accepter le site client
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Structure du code}
  \begin{itemize}
    \bitem \texttt{server/main.py} : point d'entrée, configuration
    \bitem \texttt{server/api.py} : routes et endpoints
    \bitem \texttt{server/schemas.py} : modèles Pydantic
    \bitem \texttt{server/metrics.py} : métriques Prometheus
    \bitem \texttt{server/virus\_total.py} : scan VirusTotal
    \bitem \texttt{server/glpi.py} : création tickets GLPI
  \end{itemize}
\end{frame}
\note{
  Le serveur est le cœur du système : il reçoit les données de tous les agents.
  FastAPI a été choisi pour ses performances (asynchrone) et sa facilité de développement.
  L'authentification est simple mais efficace : une API key partagée, changée en production.
}

% --- Slide 9 : EDR Server — Intégrations ---
\begin{frame}{EDR Server — VirusTotal \& GLPI}
  \begin{block}{VirusTotal — Scan automatique}
    \begin{itemize}
      \bitem Quand un agent envoie un fichier suspect, le serveur le transmet à l'API VirusTotal
      \bitem \textbf{Worker asynchrone} : ne bloque pas le thread principal
      \bitem Résultat : score de réputation, moteurs de détection déclenchés
      \bitem Si score $>$ seuil $\rightarrow$ alerte immédiate
    \end{itemize}
  \end{block}
  \vspace{0.3cm}
  \begin{block}{GLPI — Gestion de tickets automatique}
    \begin{itemize}
      \bitem Quand un événement \textbf{CRITICAL} est détecté $\rightarrow$ création automatique d'un ticket
      \bitem Le ticket contient : type de menace, agent concerné, horodatage, hash du fichier
      \bitem L'utilisateur voit le ticket dans son portail client en temps réel
      \bitem OAuth2 pour l'authentification entre EDR et GLPI
    \end{itemize}
  \end{block}
\end{frame}
\note{
  VirusTotal agrège les résultats de 70+ moteurs antivirus.
  Si le fichier est déjà connu, VT répond en quelques secondes (cache).
  GLPI = solution open-source de ticketing IT. On l'utilise comme back-office.
  L'intégration est complète : détection $\rightarrow$ ticket $\rightarrow$ visible dans le portail.
}

% --- Slide 10 : EDR Server — Base de Données ---
\begin{frame}{EDR Server — Persistance}
  \textbf{PostgreSQL comme moteur principal}
  \begin{itemize}
    \bitem Choix : \textbf{PostgreSQL} pour la fiabilité, les performances et les fonctionnalités avancées
    \bitem \textbf{Tables principales} :
    \begin{itemize}
      \bitem \texttt{agents} : hostname, OS, version, IP, last\_seen, statut (actif/silencieux/inactif)
      \bitem \texttt{events} : type, sévérité (info/warning/critical), timestamp, agent concerné
      \bitem \texttt{scans} : hash du fichier, score VirusTotal, statut (pending/done/error)
    \end{itemize}
    \bitem \textbf{SQLite} disponible en option pour déploiement léger (dev, tests)
  \end{itemize}
\end{frame}
\note{
  On utilise PostgreSQL en production car c'est plus robuste pour la concurrence et les données critiques.
  SQLite est gardé pour le développement local — plus simple à setup.
  Les schémas sont versionnés dans le dépôt Git.
}

% --- Slide 11 : EDR Agent ---
\begin{frame}{EDR Agent — Architecture}
  \textbf{Un agent Python compilé en binaire standalone}
  \begin{itemize}
    \bitem Compilation via \textbf{PyInstaller} : un seul binaire, pas de dépendance Python requise
    \bitem Compatible \textbf{Linux, macOS, Windows}
    \bitem Taille : $\sim$15 Mo, léger et non-intrusif
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Modules internes}
  \begin{itemize}
    \bitem \texttt{main.py} : orchestration, boucle principale
    \bitem \texttt{heartbeat.py} : signal de vie toutes les 30s
    \bitem \texttt{sender.py} : communication HTTP avec le serveur
    \bitem \texttt{fim.py} : File Integrity Monitoring
    \bitem \texttt{collector.py} : collecte des informations système
    \bitem \texttt{hasher.py} : calcul de hash (SHA-256)
    \bitem \texttt{reporter.py} : génération de rapports
    \bitem \texttt{vt\_checker.py} : vérification VirusTotal
  \end{itemize}
\end{frame}
\note{
  L'agent est conçu pour être déployé facilement sur n'importe quel poste.
  PyInstaller permet de livrer un binaire sans que l'utilisateur ait à installer Python.
  Le binaire est signé et vérifié avant exécution.
}

% --- Slide 12 : EDR Agent — Fonctionnalités ---
\begin{frame}{EDR Agent — Fonctionnalités Clés}
  \begin{columns}[t]
    \column{0.48\textwidth}
    \textbf{Scan de fichiers}
    \begin{itemize}
      \bitem Analyse des processus en cours
      \bitem Calcul de hash des fichiers suspects
      \bitem Vérification réputation VirusTotal
      \bitem Détection de connexions réseau anormales
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Mode monitor (daemon)}
    \begin{itemize}
      \bitem Fonctionne en continu en arrière-plan
      \bitem Surveillance temps réel du système
      \bitem Envoi heartbeat toutes les 30s
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{FIM — File Integrity Monitoring}
    \begin{itemize}
      \bitem Détecte les modifications de fichiers critiques (/etc/passwd, binaires systèmes)
      \bitem Utilise SHA-256 pour vérifier l'intégrité
      \bitem Alerte si un fichier surveillé est modifié
    \end{itemize}
    \vspace{0.3cm}
    \textbf{Déploiement}
    \begin{itemize}
      \bitem \texttt{curl -sSL ... | sudo bash}
      \bitem Auto-détection de l'OS et de l'architecture
      \bitem Installation en $\sim$10 secondes
      \bitem Auto-update via heartbeat
    \end{itemize}
  \end{columns}
\end{frame}
\note{
  L'agent a deux modes : scan (une analyse ponctuelle) et monitor (surveillance continue).
  Le FIM est particulièrement important pour la sécurité — il détecte si un attaquant modifie des fichiers système.
  L'installation one-liner est conçue pour être la plus simple possible : une commande, pas de questions.
}

% --- Slide 13 : Communication Agent-Serveur ---
\begin{frame}{Communication Agent $\leftrightarrow$ Serveur}
  \begin{itemize}
    \bitem \textbf{Protocole} : HTTP POST, payload JSON
    \bitem \textbf{Authentification} : API key dans le header \texttt{X-API-Key}
    \bitem \textbf{Payload} : type d'événement, données, horodatage, hash
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Cycle de vie d'un événement}
  \begin{enumerate}
    \item Agent détecte un fichier suspect $\rightarrow$ calcule le hash SHA-256
    \item Envoie au serveur EDR via API REST
    \item Serveur vérifie le hash : déjà vu ? connu de VirusTotal ?
    \item Si nouveau : scan VirusTotal asynchrone
    \item Résultat : bénin → ignoré / suspect → alerte / critique → ticket GLPI
    \item Agent reçoit la réponse et agit (alerte locale, quarantaine)
  \end{enumerate}
\end{frame}
\note{
  La communication est simple mais efficace : HTTP REST avec JSON.
  L'agent attend une réponse du serveur pour chaque événement :
  - "OK" : événement enregistré
  - "ALERT" : menace détectée, l'agent doit agir
  - "QUARANTINE" : le fichier doit être mis en quarantaine
}

% --- Slide 15 : Site Client — Authentification & Base ---
\begin{frame}{Site Client — Authentification \& Base de Données}
  \textbf{Authentification sécurisée}
  \begin{itemize}
    \bitem Inscription avec email + mot de passe (haché en bcrypt)
    \bitem Connexion sécurisée, session PHP, protection CSRF
    \bitem Rôles : \textbf{client} (accès à ses données) / \textbf{admin} (gestion complète)
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Base MySQL}
  \begin{itemize}
    \bitem Table \texttt{clients} : nom, email, mot de passe (hashé), abonnement, date création

    \bitem Table \texttt{metrics} : connexions, téléchargements, actions
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Sécurité des données}
  \begin{itemize}
    \bitem Connexion BDD sécurisée, requêtes préparées (PDO)
    \bitem Pas de mot de passe en clair, pas de données sensibles exposées
    \bitem Validation des entrées côté serveur
  \end{itemize}
\end{frame}
\note{
  La sécurité est primordiale : mots de passe hachés avec bcrypt, requêtes préparées pour éviter les injections SQL.
  Le site ne stocke que les données nécessaires : le reste est dans GLPI.
  Les sessions PHP sont sécurisées avec des tokens CSRF.
}

% --- Slide 16 : Site Client — GLPI SSO ---
\begin{frame}{Site Client — GLPI SSO \& Synchronisation}
  \textbf{Intégration transparente avec GLPI}
  \begin{itemize}
    \bitem \textbf{OAuth2} : connexion SSO via GLPI
    \bitem L'utilisateur se connecte au site avec son compte GLPI
    \bitem Pas de double authentification — une seule session pour tout
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Synchronisation des tickets}
  \begin{itemize}
    \bitem Les tickets créés par l'EDR (événements critiques) sont visibles dans le portail
    \bitem Synchronisation automatique : création, mise à jour, statut
    \bitem Le client peut commenter ses tickets directement depuis le portail
    \bitem Les commentaires sont synchronisés avec GLPI
  \end{itemize}
  \vspace{0.3cm}
  \begin{exampleblock}{Parcours type}
    Détection EDR $\rightarrow$ Ticket GLPI $\rightarrow$ Visible dans le portail $\rightarrow$ Le client suit l'incident en temps réel
  \end{exampleblock}
\end{frame}
\note{
  Le SSO simplifie l'expérience utilisateur : un seul compte pour GLPI et le portail.
  Les tickets sont synchronisés en quasi-temps réel via l'API GLPI.
  Le client n'a pas besoin d'accéder à GLPI directement — tout est dans le portail.
}

\begin{frame}{Chatbot — Architecture}
  \textbf{Un assistant virtuel pour les clients LockBits}
  \begin{itemize}
    \bitem \textbf{Proxy PHP} vers l'API Claude (Anthropic)
    \bitem Point d'entrée unique : \texttt{chat.php}
    \bitem Intégré au site client via une fenêtre de chat dédiée
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Fonctionnalités}
  \begin{itemize}
    \bitem Répond aux questions sur LockBits : services, abonnement, utilisation
    \bitem Aide au diagnostic : interprète les alertes EDR pour le client
    \bitem Historique de conversation conservé pendant la session
    \bitem Réponses en langage naturel grâce à Claude API
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Fonctionnement}
  \begin{enumerate}
    \item L'utilisateur envoie un message $\rightarrow$ PHP le valide et le nettoie
    \item PHP forwarde la requête à Claude API avec un \textbf{system prompt} précis
    \item Claude répond $\rightarrow$ PHP vérifie la réponse $\rightarrow$ l'affiche dans le chat
  \end{enumerate}
\end{frame}
\note{
  Le chatbot est conçu pour assister les clients sans solliciter l'équipe technique.
  Il utilise Claude API d'Anthropic pour générer des réponses naturelles et pertinentes.
  Le system prompt limite le chatbot au contexte LockBits — il ne répond pas à des questions hors-sujet.
}

% --- Slide 19 : Chatbot — Sécurité ---
\begin{frame}{Chatbot — Sécurité \& Limitation}
  \textbf{Protection contre les abus}
  \begin{itemize}
    \bitem \textbf{Rate limiting} : compteur quotidien par IP, stocké dans fichier JSON
    \bitem Limite : \textbf{100 requêtes par jour} par utilisateur
    \bitem Réinitialisation automatique chaque jour
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Sécurité des échanges}
  \begin{itemize}
    \bitem \textbf{Clé API Claude} stockée dans \texttt{.env} — jamais exposée au frontend
    \bitem \textbf{Validation des entrées} : pas de code exécutable, pas d'injection
    \bitem \textbf{System prompt verrouillé} : le chatbot ne répond que sur le périmètre LockBits
    \bitem \textbf{Aucune donnée sensible} transmise à Claude (les messages sont anonymisés)
  \end{itemize}
  \vspace{0.3cm}
  \textbf{Déploiement}
  \begin{itemize}
    \bitem Fichier unique \texttt{chat.php}, léger et rapide
    \bitem Dockerisé avec le site client (même conteneur)
    \bitem Accessible via \texttt{chat.lockbits.pro}
  \end{itemize}
\end{frame}
\note{
  Le rate limiting est important pour éviter les abus et maîtriser les coûts de l'API Claude.
  Le system prompt est conçu pour que le chatbot reste dans son rôle : assistant LockBits.
  Aucune donnée personnelle ou sensible n'est envoyée à Claude — les messages sont anonymisés côté PHP.
}
